import random
from treys import Card, Evaluator

BOT_ID = "BOT_TESTE"
SALDO_MIN_REBUY = 200.0
SALDO_MAX_REBUY = 10000.0

TRADUCAO_MAOS = {
    "Royal Flush": "Royal Flush",
    "Straight Flush": "Straight Flush",
    "Four of a Kind": "Quadra",
    "Full House": "Full House",
    "Flush": "Flush",
    "Straight": "Sequência",
    "Three of a Kind": "Trinca",
    "Two Pair": "Dois Pares",
    "Pair": "Um Par",
    "High Card": "Carta Alta",
}

PASSOS_SMALL_BLIND = [5.0, 10.0, 20.0, 25.0, 50.0, 100.0, 200.0, 500.0]


def fmt_valor(v: float) -> str:
    v = round(v, 2)
    return f"{v:.0f}" if v == int(v) else f"{v:.2f}"


class JogoPoker:
    def __init__(self):
        self.evaluator = Evaluator()

        # Jogadores atualmente conectados (+ Bot se ativo)
        self.jogadores: dict = {}
        self.ordem_jogadores: list = []

        # Histórico financeiro de jogadores que saíram após jogar (para o placar final)
        self.historico_desconectados: list[dict] = []

        # Configurações da mesa
        self.indice_sb_cfg = 1  # Padrão: SB=10 / BB=20
        self.small_blind = PASSOS_SMALL_BLIND[self.indice_sb_cfg]
        self.big_blind = self.small_blind * 2.0
        self.bot_ativo = False

        # Estado da sessão / rodada
        self.fase = "AGUARDANDO"
        self.numero_rodada = 0
        self.indice_dealer_ordem = -1
        self.participantes_mao: list = []
        self.dealer_id = None
        self.sb_id = None
        self.bb_id = None

        # Controle de scanner e cartas
        self.cartas_usadas_rodada: set[str] = set()
        self.buffer_cartas: list[str] = []
        self.indice_distribuicao = 0
        self.cartas_mesa_ocultas: list[str] = []
        self.cartas_mesa_reveladas: list[str] = []
        self.aviso_scanner = ""

        # Controle de apostas e turnos
        self.pote = 0.0
        self.aposta_atual = 0.0
        self.indice_turno_mao = 0
        self.agiram_na_street: set = set()
        self.ultima_acao_msg = "Aguardando jogadores conectarem..."
        self.resultado_rodada: list[dict] = []
        self.placar_final: list[dict] = []

    # =========================================================================
    # CONFIGURAÇÃO NO LOBBY (BLINDS, REORDENAÇÃO, BOT DE TESTES)
    # =========================================================================
    def ajustar_blinds(self, delta: int) -> None:
        if self.fase != "AGUARDANDO":
            return
        novo = max(0, min(len(PASSOS_SMALL_BLIND) - 1, self.indice_sb_cfg + delta))
        self.indice_sb_cfg = novo
        self.small_blind = PASSOS_SMALL_BLIND[novo]
        self.big_blind = self.small_blind * 2.0

    def alternar_bot_testes(self) -> list[tuple]:
        if self.fase != "AGUARDANDO":
            return []
        if self.bot_ativo:
            self.bot_ativo = False
            self.jogadores.pop(BOT_ID, None)
            if BOT_ID in self.ordem_jogadores:
                self.ordem_jogadores.remove(BOT_ID)
            self.ultima_acao_msg = "Modo Testes (Bot) desativado."
        else:
            self.bot_ativo = True
            self.jogadores[BOT_ID] = {
                "id": BOT_ID,
                "nome": "Robô (Bot)",
                "is_bot": True,
                "saldo": 1000.0,
                "saldo_inicial": 1000.0,
                "total_investido": 1000.0,
                "mao": [],
                "aposta_rodada": 0.0,
                "contribuicao_mao": 0.0,
                "ativo": False,
                "all_in": False,
                "na_mao": False,
                "em_espera": False,
                "jogou_sessao": False,
            }
            if BOT_ID not in self.ordem_jogadores:
                self.ordem_jogadores.append(BOT_ID)
            self.ultima_acao_msg = "Modo Testes ativado: Robô (Bot) entrou na mesa."
        return self._gerar_estado_individualizado(self.ultima_acao_msg)

    def mover_jogador(self, jid, direcao: int) -> list[tuple]:
        """direcao = -1 (subir) ou +1 (descer) na ordem física da mesa."""
        if self.fase != "AGUARDANDO" or jid not in self.ordem_jogadores:
            return []
        idx = self.ordem_jogadores.index(jid)
        novo_idx = idx + direcao
        if 0 <= novo_idx < len(self.ordem_jogadores):
            self.ordem_jogadores[idx], self.ordem_jogadores[novo_idx] = (
                self.ordem_jogadores[novo_idx],
                self.ordem_jogadores[idx],
            )
        return self._gerar_estado_individualizado("Ordem da mesa atualizada.")

    # =========================================================================
    # CONEXÃO, DESCONEXÃO E REBUY
    # =========================================================================
    def registrar_jogador(self, conn, nome: str, saldo: float) -> list[tuple]:
        saldo_f = round(float(saldo), 2)
        em_espera = self.fase != "AGUARDANDO"

        self.jogadores[conn] = {
            "id": conn,
            "nome": nome,
            "is_bot": False,
            "saldo": saldo_f,
            "saldo_inicial": saldo_f,
            "total_investido": saldo_f,
            "mao": [],
            "aposta_rodada": 0.0,
            "contribuicao_mao": 0.0,
            "ativo": False,
            "all_in": False,
            "na_mao": False,
            "em_espera": em_espera,
            "jogou_sessao": False,
        }
        if conn not in self.ordem_jogadores:
            self.ordem_jogadores.append(conn)

        mensagens = []
        if em_espera:
            for c in self.cartas_mesa_reveladas:
                mensagens.append((conn, "carta_mesa", {"carta": c}))
            if self.fase == "PLACAR_FINAL":
                mensagens.append((conn, "placar_final", {"ranking": self.placar_final}))
            aviso = f"{nome} entrou em modo de espera (jogará na próxima rodada)."
        else:
            aviso = f"{nome} conectou com R$ {fmt_valor(saldo_f)}."

        self.ultima_acao_msg = aviso
        mensagens.extend(self._gerar_estado_individualizado(aviso))
        return mensagens

    def remover_jogador(self, conn) -> list[tuple]:
        jogador = self.jogadores.pop(conn, None)
        if not jogador:
            return []

        if conn in self.ordem_jogadores:
            self.ordem_jogadores.remove(conn)

        if jogador["jogou_sessao"]:
            self.historico_desconectados.append(
                {
                    "nome": f"{jogador['nome']} (saiu)",
                    "total_investido": jogador["total_investido"],
                    "saldo_final": jogador["saldo"],
                    "lucro": round(jogador["saldo"] - jogador["total_investido"], 2),
                }
            )

        nome = jogador["nome"]
        mensagens = []

        # Se desconectou durante a distribuição de cartas privadas
        if self.fase == "DISTRIBUINDO_MAOS" and conn in self.participantes_mao:
            alvo_atual = self.jogador_alvo_distribuicao()
            if alvo_atual == conn:
                for c in self.buffer_cartas:
                    self.cartas_usadas_rodada.discard(c)
                self.buffer_cartas.clear()
            idx_rem = self.participantes_mao.index(conn)
            self.participantes_mao.remove(conn)
            if idx_rem < self.indice_distribuicao:
                self.indice_distribuicao -= 1

            if len(self.participantes_mao) < 2:
                self.fase = "AGUARDANDO"
                self.ultima_acao_msg = f"{nome} saiu. Jogadores insuficientes; voltando ao lobby."
                mensagens.append(("TODOS", "limpar_mesa", {}))
                mensagens.extend(self._gerar_estado_individualizado(self.ultima_acao_msg))
                return mensagens

            if self.indice_distribuicao >= len(self.participantes_mao):
                self.fase = "DISTRIBUINDO_MESA"
                self.buffer_cartas.clear()

        # Se desconectou durante uma rodada de apostas
        elif self.fase in ("PRE_FLOP", "FLOP", "TURN", "RIVER") and conn in self.participantes_mao:
            jogador["ativo"] = False
            era_vez = (
                self.participantes_mao
                and self.participantes_mao[self.indice_turno_mao] == conn
            )
            self.agiram_na_street.add(conn)
            ativos = self._ativos_na_mao()

            if len(ativos) == 1:
                vencedor_id = ativos[0]
                vencedor = self.jogadores[vencedor_id]
                premio = round(self.pote, 2)
                vencedor["saldo"] = round(vencedor["saldo"] + premio, 2)
                self.resultado_rodada = [
                    {
                        "pote_nome": "Pote Principal",
                        "valor": premio,
                        "vencedores": [vencedor["nome"]],
                        "premio_cada": premio,
                        "combinacao": "Oponente desconectou",
                    }
                ]
                self.fase = "SHOWDOWN"
                self.ultima_acao_msg = (
                    f"{nome} desconectou. {vencedor['nome']} ganhou R$ {fmt_valor(premio)}!"
                )
                return self._gerar_estado_individualizado(self.ultima_acao_msg)

            if self._street_encerrada():
                return self._concluir_street(f"{nome} desconectou.")

            if era_vez:
                self._avancar_turno()

        self.ultima_acao_msg = f"{nome} saiu da mesa."
        mensagens.extend(self._gerar_estado_individualizado(self.ultima_acao_msg))
        return mensagens

    def processar_rebuy(self, conn, valor: float) -> list[tuple]:
        jogador = self.jogadores.get(conn)
        if not jogador:
            return []

        if self.fase not in ("AGUARDANDO", "SHOWDOWN") and jogador["na_mao"]:
            return [
                (
                    conn,
                    "erro",
                    {"msg": "Aguarde o fim da rodada atual para fazer Rebuy."},
                )
            ]

        valor_f = round(float(valor), 2)
        if valor_f < SALDO_MIN_REBUY or valor_f > SALDO_MAX_REBUY:
            return [
                (
                    conn,
                    "erro",
                    {
                        "msg": f"Rebuy deve ser entre R$ {fmt_valor(SALDO_MIN_REBUY)} e R$ {fmt_valor(SALDO_MAX_REBUY)}."
                    },
                )
            ]

        jogador["saldo"] = round(jogador["saldo"] + valor_f, 2)
        jogador["total_investido"] = round(jogador["total_investido"] + valor_f, 2)
        self.ultima_acao_msg = (
            f"{jogador['nome']} fez Rebuy de R$ {fmt_valor(valor_f)} "
            f"(Saldo: R$ {fmt_valor(jogador['saldo'])})."
        )
        return self._gerar_estado_individualizado(self.ultima_acao_msg)

    # =========================================================================
    # INÍCIO DE PARTIDA / NOVA RODADA
    # =========================================================================
    def pode_iniciar_rodada(self) -> tuple[bool, str]:
        aptos = [
            jid
            for jid in self.ordem_jogadores
            if self.jogadores[jid]["saldo"] > 0
            or (self.jogadores[jid]["is_bot"] and self.bot_ativo)
        ]
        if len(aptos) < 2:
            return False, "Necessário pelo menos 2 jogadores com saldo > R$ 0."
        return True, ""

    def iniciar_nova_rodada(self) -> list[tuple]:
        # Rebuy automático do Bot se zerou para não travar os testes
        if self.bot_ativo and BOT_ID in self.jogadores:
            bot = self.jogadores[BOT_ID]
            if bot["saldo"] <= 0:
                bot["saldo"] = 1000.0
                bot["total_investido"] = round(bot["total_investido"] + 1000.0, 2)

        aptos = [
            jid
            for jid in self.ordem_jogadores
            if self.jogadores[jid]["saldo"] > 0
        ]
        if len(aptos) < 2:
            self.ultima_acao_msg = "Mínimo 2 jogadores com saldo > R$ 0 para iniciar."
            return self._gerar_estado_individualizado(self.ultima_acao_msg)

        self.numero_rodada += 1
        self.cartas_usadas_rodada.clear()
        self.buffer_cartas.clear()
        self.cartas_mesa_ocultas.clear()
        self.cartas_mesa_reveladas.clear()
        self.resultado_rodada.clear()
        self.aviso_scanner = ""
        self.pote = 0.0
        self.aposta_atual = 0.0
        self.agiram_na_street.clear()

        for jid in self.ordem_jogadores:
            j = self.jogadores[jid]
            j["mao"].clear()
            j["aposta_rodada"] = 0.0
            j["contribuicao_mao"] = 0.0
            j["all_in"] = False
            if jid in aptos:
                j["na_mao"] = True
                j["ativo"] = True
                j["em_espera"] = False
                j["jogou_sessao"] = True
            else:
                j["na_mao"] = False
                j["ativo"] = False

        # Rotaciona o Dealer entre os participantes aptos
        if self.dealer_id in aptos:
            idx_d = (aptos.index(self.dealer_id) + 1) % len(aptos)
        else:
            idx_d = 0

        self.dealer_id = aptos[idx_d]

        # Reordena a lista da mão começando no Dealer (índice 0 = Dealer)
        self.participantes_mao = aptos[idx_d:] + aptos[:idx_d]

        # Define Small Blind e Big Blind (Heads-Up vs 3+ jogadores)
        n = len(self.participantes_mao)
        if n == 2:
            # Heads-Up: Dealer é SB, outro é BB
            self.sb_id = self.participantes_mao[0]
            self.bb_id = self.participantes_mao[1]
            # Ordem de entrega de cartas: começa pelo BB ou SB (vamos iniciar pelo SB / ordem da mesa)
            self.ordem_distribuicao = [self.sb_id, self.bb_id]
        else:
            self.sb_id = self.participantes_mao[1]
            self.bb_id = self.participantes_mao[2]
            # Começa à esquerda do Dealer (SB, BB, ..., Dealer)
            self.ordem_distribuicao = self.participantes_mao[1:] + [
                self.participantes_mao[0]
            ]

        self.indice_distribuicao = 0
        self.fase = "DISTRIBUINDO_MAOS"
        nome_primeiro = self.jogadores[self.ordem_distribuicao[0]]["nome"]
        self.ultima_acao_msg = f"Rodada #{self.numero_rodada}: escaneie 2 cartas para {nome_primeiro}."

        mensagens = [("TODOS", "limpar_mesa", {})]
        mensagens.extend(self._gerar_estado_individualizado(self.ultima_acao_msg))
        return mensagens

    # =========================================================================
    # LEITURA DO SCANNER E CONFIRMAÇÕES DO DEALER
    # =========================================================================
    def jogador_alvo_distribuicao(self):
        if self.fase != "DISTRIBUINDO_MAOS":
            return None
        if 0 <= self.indice_distribuicao < len(self.ordem_distribuicao):
            return self.ordem_distribuicao[self.indice_distribuicao]
        return None

    def processar_carta_lida(self, carta: str) -> list[tuple]:
        if self.fase not in ("DISTRIBUINDO_MAOS", "DISTRIBUINDO_MESA"):
            return []

        if carta in self.cartas_usadas_rodada:
            self.aviso_scanner = "Carta repetida bloqueada! Passe outra carta."
            return []

        limite = 2 if self.fase == "DISTRIBUINDO_MAOS" else 5
        if len(self.buffer_cartas) >= limite:
            self.aviso_scanner = (
                f"Limite {limite}/{limite} atingido. Clique em Confirmar ou Limpar."
            )
            return []

        self.buffer_cartas.append(carta)
        self.cartas_usadas_rodada.add(carta)
        self.aviso_scanner = f"Carta {len(self.buffer_cartas)}/{limite} registrada."
        return []

    def registrar_aviso_duplicada(self) -> None:
        if self.fase in ("DISTRIBUINDO_MAOS", "DISTRIBUINDO_MESA"):
            self.aviso_scanner = "Carta já usada nesta rodada! Troque a carta."

    def limpar_buffer_atual(self) -> None:
        if self.fase not in ("DISTRIBUINDO_MAOS", "DISTRIBUINDO_MESA"):
            return
        for c in self.buffer_cartas:
            self.cartas_usadas_rodada.discard(c)
        self.buffer_cartas.clear()
        self.aviso_scanner = "Leitura limpa. Escaneie novamente."

    def confirmar_distribuicao_atual(self) -> list[tuple]:
        mensagens = []

        if self.fase == "DISTRIBUINDO_MAOS":
            if len(self.buffer_cartas) != 2:
                self.aviso_scanner = "Escaneie as 2 cartas antes de confirmar."
                return []

            alvo_id = self.jogador_alvo_distribuicao()
            if alvo_id is None or alvo_id not in self.jogadores:
                return []

            jogador = self.jogadores[alvo_id]
            jogador["mao"] = list(self.buffer_cartas)
            self.buffer_cartas.clear()
            self.aviso_scanner = ""

            # Envia cartas privadas apenas ao dono (se humano)
            if not jogador["is_bot"]:
                for c in jogador["mao"]:
                    mensagens.append((alvo_id, "carta_mao", {"carta": c}))

            self.indice_distribuicao += 1
            if self.indice_distribuicao < len(self.ordem_distribuicao):
                prox_nome = self.jogadores[
                    self.ordem_distribuicao[self.indice_distribuicao]
                ]["nome"]
                self.ultima_acao_msg = f"Cartas entregues. Próximo: {prox_nome}."
            else:
                self.fase = "DISTRIBUINDO_MESA"
                self.ultima_acao_msg = (
                    "Mãos entregues! Escaneie as 5 cartas da mesa (viradas para baixo)."
                )

            mensagens.extend(self._gerar_estado_individualizado(self.ultima_acao_msg))
            return mensagens

        if self.fase == "DISTRIBUINDO_MESA":
            if len(self.buffer_cartas) != 5:
                self.aviso_scanner = "Escaneie todas as 5 cartas da mesa antes de confirmar."
                return []

            # Guarda as 5 cartas viradas para baixo (NÃO envia aos celulares)
            self.cartas_mesa_ocultas = list(self.buffer_cartas)
            self.cartas_mesa_reveladas.clear()
            self.buffer_cartas.clear()
            self.aviso_scanner = ""

            return self._iniciar_preflop_com_blinds()

        return []

    def _cobrar_blind(self, jid, valor_blind: float) -> float:
        j = self.jogadores[jid]
        pago = round(min(valor_blind, j["saldo"]), 2)
        j["saldo"] = round(j["saldo"] - pago, 2)
        j["aposta_rodada"] = round(j["aposta_rodada"] + pago, 2)
        j["contribuicao_mao"] = round(j["contribuicao_mao"] + pago, 2)
        self.pote = round(self.pote + pago, 2)
        if j["saldo"] <= 0:
            j["saldo"] = 0.0
            j["all_in"] = True
        return pago

    def _iniciar_preflop_com_blinds(self) -> list[tuple]:
        self.fase = "PRE_FLOP"
        self.agiram_na_street.clear()
        for jid in self.participantes_mao:
            self.jogadores[jid]["aposta_rodada"] = 0.0

        pago_sb = self._cobrar_blind(self.sb_id, self.small_blind)
        pago_bb = self._cobrar_blind(self.bb_id, self.big_blind)
        self.aposta_atual = max(pago_sb, pago_bb)

        sb_nome = self.jogadores[self.sb_id]["nome"]
        bb_nome = self.jogadores[self.bb_id]["nome"]
        msg_blinds = (
            f"Blinds: {sb_nome} (SB R${fmt_valor(pago_sb)}), "
            f"{bb_nome} (BB R${fmt_valor(pago_bb)})."
        )

        # Se menos de 2 jogadores podem apostar (all-in já nos blinds), vai virar o Flop
        if len(self._ativos_que_podem_agir()) <= 1 and self._todos_igualaram_ou_allin():
            return self._concluir_street(msg_blinds)

        n = len(self.participantes_mao)
        if n == 2:
            # Heads-Up Pré-Flop: Dealer/SB (índice 0) age primeiro
            inicio = 0
        else:
            # 3+ jogadores Pré-Flop: UTG (índice 3 % n) age primeiro
            inicio = 3 % n

        self.indice_turno_mao = inicio
        if not self._pode_agir(self.participantes_mao[self.indice_turno_mao]):
            self._avancar_turno()

        self.ultima_acao_msg = msg_blinds
        return self._gerar_estado_individualizado(msg_blinds)

    # =========================================================================
    # VIRAR CARTAS COMUNITÁRIAS (CONFIRMAÇÃO DO DEALER)
    # =========================================================================
    def confirmar_virar_cartas_mesa(self) -> list[tuple]:
        mapa = {
            "AGUARDANDO_VIRAR_FLOP": ("FLOP", 3, "Flop virado!"),
            "AGUARDANDO_VIRAR_TURN": ("TURN", 4, "Turn virado!"),
            "AGUARDANDO_VIRAR_RIVER": ("RIVER", 5, "River virado!"),
        }
        if self.fase not in mapa:
            return []

        proxima_fase, total_reveladas, titulo = mapa[self.fase]
        novas_cartas = self.cartas_mesa_ocultas[
            len(self.cartas_mesa_reveladas) : total_reveladas
        ]
        self.cartas_mesa_reveladas.extend(novas_cartas)

        mensagens = []
        for c in novas_cartas:
            mensagens.append(("TODOS", "carta_mesa", {"carta": c}))

        self.fase = proxima_fase
        self.aposta_atual = 0.0
        self.agiram_na_street.clear()
        for jid in self.participantes_mao:
            if jid in self.jogadores:
                self.jogadores[jid]["aposta_rodada"] = 0.0

        # Em cenário de All-in (menos de 2 jogadores com fichas para apostar),
        # avança para pedir ao dealer virar a próxima fase ou ir ao Showdown
        if len(self._ativos_que_podem_agir()) < 2:
            mensagens.extend(self._concluir_street(f"{titulo} (All-in na mesa)"))
            return mensagens

        # Pós-flop: primeiro jogador ativo à esquerda do Dealer (índice 1 em diante)
        self.indice_turno_mao = 1 % len(self.participantes_mao)
        if not self._pode_agir(self.participantes_mao[self.indice_turno_mao]):
            self._avancar_turno()

        self.ultima_acao_msg = f"{titulo} Rodada de apostas."
        mensagens.extend(self._gerar_estado_individualizado(self.ultima_acao_msg))
        return mensagens

    # =========================================================================
    # PROCESSAMENTO DE AÇÕES (FOLD, CHECK, CALL, RAISE, ALL-IN)
    # =========================================================================
    def _pode_agir(self, jid) -> bool:
        j = self.jogadores.get(jid)
        return bool(j and j["na_mao"] and j["ativo"] and not j["all_in"] and j["saldo"] > 0)

    def _ativos_na_mao(self) -> list:
        return [
            jid
            for jid in self.participantes_mao
            if jid in self.jogadores and self.jogadores[jid]["ativo"]
        ]

    def _ativos_que_podem_agir(self) -> list:
        return [jid for jid in self.participantes_mao if self._pode_agir(jid)]

    def _todos_igualaram_ou_allin(self) -> bool:
        for jid in self._ativos_na_mao():
            j = self.jogadores[jid]
            if not j["all_in"] and j["aposta_rodada"] < self.aposta_atual:
                return False
        return True

    def _street_encerrada(self) -> bool:
        podem_agir = self._ativos_que_podem_agir()
        if not podem_agir:
            return True
        if not self._todos_igualaram_ou_allin():
            return False
        return all(jid in self.agiram_na_street for jid in podem_agir)

    def jogador_da_vez_id(self):
        if self.fase not in ("PRE_FLOP", "FLOP", "TURN", "RIVER"):
            return None
        if not self.participantes_mao:
            return None
        jid = self.participantes_mao[self.indice_turno_mao]
        return jid if self._pode_agir(jid) else None

    def jogador_da_vez_eh_bot(self) -> bool:
        return self.jogador_da_vez_id() == BOT_ID

    def processar_acao(self, conn, acao: str, valor: float = 0.0) -> list[tuple]:
        if self.fase not in ("PRE_FLOP", "FLOP", "TURN", "RIVER"):
            return []
        if conn not in self.jogadores:
            return []
        if self.jogador_da_vez_id() != conn:
            return [(conn, "erro", {"msg": "Não é a sua vez!"})]

        j = self.jogadores[conn]
        nome = j["nome"]
        a_pagar = round(max(0.0, self.aposta_atual - j["aposta_rodada"]), 2)
        aviso = ""

        if acao == "fold":
            j["ativo"] = False
            self.agiram_na_street.add(conn)
            aviso = f"{nome} correu (Fold)."

        elif acao == "check":
            if a_pagar > 0:
                return [
                    (
                        conn,
                        "erro",
                        {"msg": f"Precisa pagar R$ {fmt_valor(a_pagar)} ou correr."},
                    )
                ]
            self.agiram_na_street.add(conn)
            aviso = f"{nome} passou (Check)."

        elif acao == "call":
            pagamento = round(min(a_pagar, j["saldo"]), 2)
            j["saldo"] = round(j["saldo"] - pagamento, 2)
            j["aposta_rodada"] = round(j["aposta_rodada"] + pagamento, 2)
            j["contribuicao_mao"] = round(j["contribuicao_mao"] + pagamento, 2)
            self.pote = round(self.pote + pagamento, 2)
            if j["saldo"] <= 0:
                j["saldo"] = 0.0
                j["all_in"] = True
                aviso = f"{nome} deu ALL-IN pagando R$ {fmt_valor(pagamento)}!"
            else:
                aviso = f"{nome} pagou R$ {fmt_valor(pagamento)} (Call)."
            self.agiram_na_street.add(conn)

        elif acao == "all_in":
            pagamento = round(j["saldo"], 2)
            if pagamento <= 0:
                return [(conn, "erro", {"msg": "Saldo insuficiente."})]
            j["saldo"] = 0.0
            j["aposta_rodada"] = round(j["aposta_rodada"] + pagamento, 2)
            j["contribuicao_mao"] = round(j["contribuicao_mao"] + pagamento, 2)
            self.pote = round(self.pote + pagamento, 2)
            j["all_in"] = True
            if j["aposta_rodada"] > self.aposta_atual:
                self.aposta_atual = j["aposta_rodada"]
                self.agiram_na_street = {conn}
            else:
                self.agiram_na_street.add(conn)
            aviso = f"{nome} foi ALL-IN (R$ {fmt_valor(j['aposta_rodada'])})!"

        elif acao == "raise":
            valor_extra = round(float(valor), 2)
            total_necessario = round(a_pagar + valor_extra, 2)
            if valor_extra <= 0:
                return [(conn, "erro", {"msg": "Digite um valor maior que zero."})]
            if j["saldo"] < total_necessario:
                return [
                    (
                        conn,
                        "erro",
                        {"msg": "Saldo insuficiente para esse Raise. Use All-In."},
                    )
                ]
            j["saldo"] = round(j["saldo"] - total_necessario, 2)
            j["aposta_rodada"] = round(j["aposta_rodada"] + total_necessario, 2)
            j["contribuicao_mao"] = round(j["contribuicao_mao"] + total_necessario, 2)
            self.pote = round(self.pote + total_necessario, 2)
            self.aposta_atual = j["aposta_rodada"]
            self.agiram_na_street = {conn}
            if j["saldo"] <= 0:
                j["saldo"] = 0.0
                j["all_in"] = True
                aviso = f"{nome} aumentou em ALL-IN para R$ {fmt_valor(self.aposta_atual)}!"
            else:
                aviso = f"{nome} aumentou para R$ {fmt_valor(self.aposta_atual)} (Raise)!"
        else:
            return []

        ativos = self._ativos_na_mao()
        if len(ativos) == 1:
            vencedor_id = ativos[0]
            vencedor = self.jogadores[vencedor_id]
            premio = round(self.pote, 2)
            vencedor["saldo"] = round(vencedor["saldo"] + premio, 2)
            self.resultado_rodada = [
                {
                    "pote_nome": "Pote Principal",
                    "valor": premio,
                    "vencedores": [vencedor["nome"]],
                    "premio_cada": premio,
                    "combinacao": "Todos os outros correram",
                }
            ]
            self.fase = "SHOWDOWN"
            self.ultima_acao_msg = (
                f"{aviso} {vencedor['nome']} ganhou R$ {fmt_valor(premio)}!"
            )
            return self._gerar_estado_individualizado(self.ultima_acao_msg)

        if self._street_encerrada():
            return self._concluir_street(aviso)

        self._avancar_turno()
        self.ultima_acao_msg = aviso
        return self._gerar_estado_individualizado(aviso)

    def _avancar_turno(self) -> None:
        total = len(self.participantes_mao)
        for _ in range(total):
            self.indice_turno_mao = (self.indice_turno_mao + 1) % total
            if self._pode_agir(self.participantes_mao[self.indice_turno_mao]):
                break

    def _concluir_street(self, ultimo_aviso: str) -> list[tuple]:
        transicoes = {
            "PRE_FLOP": ("AGUARDANDO_VIRAR_FLOP", "Aguardando dealer virar o Flop (3 cartas)."),
            "FLOP": ("AGUARDANDO_VIRAR_TURN", "Aguardando dealer virar o Turn (4ª carta)."),
            "TURN": ("AGUARDANDO_VIRAR_RIVER", "Aguardando dealer virar o River (5ª carta)."),
            "RIVER": ("SHOWDOWN", ""),
        }
        nova_fase, texto = transicoes[self.fase]
        self.fase = nova_fase

        if nova_fase == "SHOWDOWN":
            return self._avaliar_showdown(ultimo_aviso)

        self.ultima_acao_msg = f"{ultimo_aviso} {texto}"
        return self._gerar_estado_individualizado(self.ultima_acao_msg)

    def calcular_potes(self) -> list[dict]:
        """Calcula Main Pot e Side Pots considerando contribuições totais na mão."""
        contribuidores_todos = [
            jid
            for jid in self.participantes_mao
            if jid in self.jogadores and self.jogadores[jid]["contribuicao_mao"] > 0
        ]
        niveis = sorted(
            {self.jogadores[jid]["contribuicao_mao"] for jid in contribuidores_todos}
        )

        potes: list[dict] = []
        nivel_ant = 0.0

        for nivel in niveis:
            delta = round(nivel - nivel_ant, 2)
            if delta <= 0:
                continue
            participam_camada = [
                jid
                for jid in contribuidores_todos
                if self.jogadores[jid]["contribuicao_mao"] >= nivel
            ]
            valor_camada = round(delta * len(participam_camada), 2)
            elegiveis = [
                jid for jid in participam_camada if self.jogadores[jid]["ativo"]
            ]

            if not elegiveis:
                if potes:
                    potes[-1]["valor"] = round(potes[-1]["valor"] + valor_camada, 2)
            elif potes and set(potes[-1]["elegiveis"]) == set(elegiveis):
                potes[-1]["valor"] = round(potes[-1]["valor"] + valor_camada, 2)
            else:
                potes.append({"valor": valor_camada, "elegiveis": elegiveis})

            nivel_ant = nivel

        return potes

    def _avaliar_showdown(self, ultimo_aviso: str) -> list[tuple]:
        board_treys = [Card.new(c) for c in self.cartas_mesa_reveladas]
        scores: dict = {}
        classes_pt: dict = {}

        for jid in self._ativos_na_mao():
            j = self.jogadores[jid]
            if len(j["mao"]) < 2:
                continue
            mao_treys = [Card.new(c) for c in j["mao"]]
            score = self.evaluator.evaluate(board_treys, mao_treys)
            classe_en = self.evaluator.class_to_string(
                self.evaluator.get_rank_class(score)
            )
            scores[jid] = score
            classes_pt[jid] = TRADUCAO_MAOS.get(classe_en, classe_en)

        potes = self.calcular_potes()
        self.resultado_rodada = []
        resumo_msgs = []
        idx_side = 0

        for i, pote_info in enumerate(potes):
            valor_pote = pote_info["valor"]
            elegiveis = [jid for jid in pote_info["elegiveis"] if jid in scores]
            if not elegiveis:
                continue

            # Excesso não pago (apenas 1 jogador contribuiu nesse nível)
            if len(elegiveis) == 1:
                jid_unico = elegiveis[0]
                self.jogadores[jid_unico]["saldo"] = round(
                    self.jogadores[jid_unico]["saldo"] + valor_pote, 2
                )
                self.resultado_rodada.append(
                    {
                        "pote_nome": "Excesso devolvido",
                        "valor": valor_pote,
                        "vencedores": [self.jogadores[jid_unico]["nome"]],
                        "premio_cada": valor_pote,
                        "combinacao": "Aposta não coberta",
                    }
                )
                continue

            melhor_score = min(scores[jid] for jid in elegiveis)
            vencedores_ids = [
                jid for jid in elegiveis if scores[jid] == melhor_score
            ]
            premio_cada = round(valor_pote / len(vencedores_ids), 2)

            # Ajuste fino de eventual centavo residual para o primeiro vencedor
            resto = round(valor_pote - premio_cada * len(vencedores_ids), 2)
            for k, vid in enumerate(vencedores_ids):
                bonus = resto if k == 0 else 0.0
                self.jogadores[vid]["saldo"] = round(
                    self.jogadores[vid]["saldo"] + premio_cada + bonus, 2
                )

            nome_pote = "Pote Principal" if i == 0 else f"Side Pot {idx_side + 1}"
            if i > 0:
                idx_side += 1

            nomes_v = [self.jogadores[vid]["nome"] for vid in vencedores_ids]
            comb = classes_pt[vencedores_ids[0]]

            self.resultado_rodada.append(
                {
                    "pote_nome": nome_pote,
                    "valor": valor_pote,
                    "vencedores": nomes_v,
                    "premio_cada": premio_cada,
                    "combinacao": comb,
                }
            )
            resumo_msgs.append(
                f"{nome_pote}: {', '.join(nomes_v)} levou R$ {fmt_valor(premio_cada)} ({comb})"
            )

        resumo_str = " | ".join(resumo_msgs) if resumo_msgs else "Rodada finalizada."
        self.ultima_acao_msg = f"SHOWDOWN! {resumo_str}"
        return self._gerar_estado_individualizado(self.ultima_acao_msg)

                mesmo_naipe = bot["mao"][0][1] == bot["mao"][1][1]

                    forca = 0.68 + (v1 / 50.0)
                else:
                    forca = ((v1 + v2) / 36.0) + (0.08 if mesmo_naipe else 0.0)

        sorteio = random.random()

        if a_pagar == 0:

        cinza_blur, 0, 255, cv2.THRESH_BINARY_INV | cv2.THRESH_OTSU
    )

    contornos, _ = cv2.findContours(binaria, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    area_max = 0.5 * recorte.size / 3
    x_min, y_min, x_max, y_max = 9999, 9999, -1, -1
    achou = False

    for c in contornos:
        a = cv2.contourArea(c)
        if a < AREA_MIN or a > area_max:
            continue
        x, y, w, h = cv2.boundingRect(c)
        x_min = min(x_min, x)
        y_min = min(y_min, y)
        x_max = max(x_max, x + w)
        y_max = max(y_max, y + h)
        achou = True

    if not achou:
        return None

    tinta = binaria[y_min:y_max, x_min:x_max]
    lado_sq = max(tinta.shape[1], tinta.shape[0])
    margem = max(2, lado_sq // 8)

    quadrado = np.zeros((lado_sq + 2 * margem, lado_sq + 2 * margem), dtype=np.uint8)
    ox = margem + (lado_sq - tinta.shape[1]) // 2
    oy = margem + (lado_sq - tinta.shape[0]) // 2
    quadrado[oy : oy + tinta.shape[0], ox : ox + tinta.shape[1]] = tinta

    img_redim = cv2.resize(quadrado, (LADO, LADO), interpolation=cv2.INTER_AREA)

    mascara = binaria[y_min:y_max, x_min:x_max]
    recorte_roi = recorte[y_min:y_max, x_min:x_max]
    media_bgr = cv2.mean(recorte_roi, mask=mascara)
    vermelho = (media_bgr[2] - 0.5 * (media_bgr[0] + media_bgr[1])) > LIMIAR_VERMELHO

    return img_redim, vermelho


def prever(modelo, nomes, img):
    caracteristicas = np.float32(calcular_hog(img))
    _, r = modelo.predict(caracteristicas)
    i = int(r[0][0])
    return nomes[i] if 0 <= i < len(nomes) else "?"

    """Cria uma mensagem padronizada no formato {'tipo': ..., 'dados': {...}}\n."""
    envelope = {"tipo": tipo, "dados": dados}
    return (json.dumps(envelope, ensure_ascii=False) + "\n").encode("utf-8")

    """Extrai (tipo, dados) de uma linha JSON recebida pelo socket."""

    return envelope.get("tipo", ""), envelope.get("dados", {})

class HardwareScanner:
    def __init__(
        self, pino_led: int, pino_vcc_ir: int, pino_do_ir: int, ativo_high: bool
    ):
        self.ativo = False
        try:
            self.led = DigitalOutputDevice(pino_led, initial_value=False)
            self.vcc_ir = DigitalOutputDevice(pino_vcc_ir, initial_value=False)
            self.sensor_ir = DigitalInputDevice(
                pino_do_ir, pull_up=None, active_state=ativo_high
            )
            self.ativo = True
            log(
                f"Hardware GPIO inicializado (LED={pino_led}, VCC_IR={pino_vcc_ir}, DO_IR={pino_do_ir}, active_high={ativo_high})."
            )
        except Exception as e:
            log(f"Hardware GPIO indisponível ({e}); operando em fallback.")

    def ligar(self) -> None:
        if self.ativo:
            self.led.on()
            self.vcc_ir.on()
            time.sleep(0.10)

    def desligar(self) -> None:
        if self.ativo:
            self.led.off()
            self.vcc_ir.off()

    @property
    def carta_no_sensor(self) -> bool:
        if not self.ativo:
            return True
        return self.sensor_ir.is_active

    def fechar(self) -> None:
        if self.ativo:
            self.desligar()
            self.led.close()
            self.vcc_ir.close()
            self.sensor_ir.close()


def orientar(bruto):

def extrair_simbolo(recorte):
    cinza = cv2.cvtColor(recorte, cv2.COLOR_BGR2GRAY)
    brilho = cv2.mean(cinza)[0]
    if brilho < LIMIAR_PRESENCA:
        return None, brilho

    cinza_blur = cv2.GaussianBlur(cinza, (3, 3), 0)
    _, binaria = cv2.threshold(
        cinza_blur, 0, 255, cv2.THRESH_BINARY_INV | cv2.THRESH_OTSU
    )

    contornos, _ = cv2.findContours(binaria, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    area_max = 0.5 * recorte.size / 3
    x_min, y_min, x_max, y_max = 9999, 9999, -1, -1
    achou = False

    for c in contornos:
        a = cv2.contourArea(c)
        if a < AREA_MIN or a > area_max:
            continue
        x, y, w, h = cv2.boundingRect(c)
        x_min = min(x_min, x)
        y_min = min(y_min, y)
        x_max = max(x_max, x + w)
        y_max = max(y_max, y + h)
        achou = True

    if not achou:
        return None, brilho

    tinta = binaria[y_min:y_max, x_min:x_max]
    lado_sq = max(tinta.shape[1], tinta.shape[0])
    margem = max(2, lado_sq // 8)

    quadrado = np.zeros((lado_sq + 2 * margem, lado_sq + 2 * margem), dtype=np.uint8)
    ox = margem + (lado_sq - tinta.shape[1]) // 2
    oy = margem + (lado_sq - tinta.shape[0]) // 2
    quadrado[oy : oy + tinta.shape[0], ox : ox + tinta.shape[1]] = tinta

    img_redim = cv2.resize(quadrado, (LADO, LADO), interpolation=cv2.INTER_AREA)

    mascara = binaria[y_min:y_max, x_min:x_max]
    recorte_roi = recorte[y_min:y_max, x_min:x_max]
    media_bgr = cv2.mean(recorte_roi, mask=mascara)
    vermelho = (media_bgr[2] - 0.5 * (media_bgr[0] + media_bgr[1])) > LIMIAR_VERMELHO

    return (img_redim, vermelho), brilho

def ler_comandos() -> None:
    for linha in sys.stdin:
        cmd = linha.strip()
        if not cmd:
            continue
        with lock_estado:
            if cmd == "LIGAR":
                estado["quer_camera"] = True
            elif cmd == "DESLIGAR":
                estado["quer_camera"] = False
            elif cmd == "RESET_LEITURA":
                estado["resetar_leitura"] = True
            elif cmd.startswith("USADAS;"):
                corpo = cmd.split(";", 1)[1].strip()
                estado["cartas_usadas"] = {
                    c.strip() for c in corpo.split(",") if c.strip()
                }
            elif cmd == "SAIR":
                estado["rodando"] = False
                break
            else:
                log(f"Comando desconhecido: {cmd}")
    with lock_estado:
        estado["rodando"] = False

                        duplicada_notificada = codigo
                        log(f"[SCANNER] Carta duplicada bloqueada: {valor} de {naipe}")
                        evento(f"DUPLICADA;{valor};{naipe}")
                else:
                    votos_duplicada = 0
                    duplicada_notificada = None
                    if leitura == candidata_atual:
                        votos += 1
                    else:
                        candidata_atual = leitura
                        votos = 1

                    if votos >= VOTOS_NECESSARIOS:
                        carta_confirmada = True
                        log(f"[SCANNER] Carta confirmada: {valor} de {naipe}")
                        evento(f"CARTA;{valor};{naipe}")
            else:
                candidata_atual = None
                votos = 0
                votos_duplicada = 0

            time.sleep(0.03)
    finally:
        if cap is not None:
            cap.stop()
            cap.close()
        hw.fechar()


if __name__ == "__main__":
    main()

NAIPES_ICONE = {"s": "♠", "h": "♥", "d": "♦", "c": "♣"}


def formatar_carta(codigo: str) -> tuple[str, str]:
    if len(codigo) < 2:
        return codigo, "#111111"
    valor = "10" if codigo[:-1].upper() == "T" else codigo[:-1].upper()
    naipe = codigo[-1].lower()
    icone = NAIPES_ICONE.get(naipe, naipe)
    cor = "#C62828" if naipe in ("h", "d") else "#111111"
    return f"{valor}{icone}", cor


class ScrollableTouchFrame(tk.Frame):
    """Container com rolagem vertical por arrasto de dedo (touch) para LCD 480x320."""

    def __init__(self, master, bg=COR_BG, **kwargs):
        super().__init__(master, bg=bg, **kwargs)
        self.canvas = tk.Canvas(self, bg=bg, bd=0, highlightthickness=0)
        self.scrollbar = tk.Scrollbar(
            self, orient="vertical", command=self.canvas.yview, width=10
        )
        self.inner = tk.Frame(self.canvas, bg=bg)
        self.win_id = self.canvas.create_window((0, 0), window=self.inner, anchor="nw")

        self.canvas.configure(yscrollcommand=self.scrollbar.set)
        self.canvas.pack(side="left", fill="both", expand=True)
        self.scrollbar.pack(side="right", fill="y")

        self.inner.bind("<Configure>", self._on_inner_configure)
        self.canvas.bind("<Configure>", self._on_canvas_configure)

        self._drag_y = 0
        self.vincular_touch(self.canvas)
        self.vincular_touch(self.inner)

    def _on_inner_configure(self, _e=None):
        self.canvas.configure(scrollregion=self.canvas.bbox("all"))

    def _on_canvas_configure(self, event):
        self.canvas.itemconfig(self.win_id, width=event.width)

    def vincular_touch(self, widget):
        widget.bind("<ButtonPress-1>", self._inicio_arrasto, add="+")
        widget.bind("<B1-Motion>", self._movimento_arrasto, add="+")

    def vincular_filhos_recursivo(self, widget):
        if not isinstance(widget, tk.Button):
            self.vincular_touch(widget)
        for filho in widget.winfo_children():
            self.vincular_filhos_recursivo(filho)

    def _inicio_arrasto(self, event):
        self._drag_y = event.y_root

    def _movimento_arrasto(self, event):
        dy = self._drag_y - event.y_root
        if abs(dy) >= 6:
            unidades = int(dy / 12)
            if unidades != 0:
                self.canvas.yview_scroll(unidades, "units")
                self._drag_y = event.y_root

    def limpar(self):
        for w in self.inner.winfo_children():
            w.destroy()
        self.canvas.yview_moveto(0)

NAIPES_ICONE = {"s": "♠", "h": "♥", "d": "♦", "c": "♣"}


def formatar_carta(codigo: str) -> tuple[str, str]:
    if len(codigo) < 2:
        return codigo, "#111111"
    valor = "10" if codigo[:-1].upper() == "T" else codigo[:-1].upper()
    naipe = codigo[-1].lower()
    icone = NAIPES_ICONE.get(naipe, naipe)
    cor = "#C62828" if naipe in ("h", "d") else "#111111"
    return f"{valor}{icone}", cor


class ScrollableTouchFrame(tk.Frame):
    """Container com rolagem vertical por arrasto de dedo (touch) para LCD 480x320."""

    def __init__(self, master, bg=COR_BG, **kwargs):
        super().__init__(master, bg=bg, **kwargs)
        self.canvas = tk.Canvas(self, bg=bg, bd=0, highlightthickness=0)
        self.scrollbar = tk.Scrollbar(
            self, orient="vertical", command=self.canvas.yview, width=10
        )
        self.inner = tk.Frame(self.canvas, bg=bg)
        self.win_id = self.canvas.create_window((0, 0), window=self.inner, anchor="nw")

        self.canvas.configure(yscrollcommand=self.scrollbar.set)
        self.canvas.pack(side="left", fill="both", expand=True)
        self.scrollbar.pack(side="right", fill="y")

        self.inner.bind("<Configure>", self._on_inner_configure)
        self.canvas.bind("<Configure>", self._on_canvas_configure)

        self._drag_y = 0
        self.vincular_touch(self.canvas)
        self.vincular_touch(self.inner)

    def _on_inner_configure(self, _e=None):
        self.canvas.configure(scrollregion=self.canvas.bbox("all"))

    def _on_canvas_configure(self, event):
        self.canvas.itemconfig(self.win_id, width=event.width)

    def vincular_touch(self, widget):
        widget.bind("<ButtonPress-1>", self._inicio_arrasto, add="+")
        widget.bind("<B1-Motion>", self._movimento_arrasto, add="+")

    def vincular_filhos_recursivo(self, widget):
        if not isinstance(widget, tk.Button):
            self.vincular_touch(widget)
        for filho in widget.winfo_children():
            self.vincular_filhos_recursivo(filho)

    def _inicio_arrasto(self, event):
        self._drag_y = event.y_root

    def _movimento_arrasto(self, event):
        dy = self._drag_y - event.y_root
        if abs(dy) >= 6:
            unidades = int(dy / 12)
            if unidades != 0:
                self.canvas.yview_scroll(unidades, "units")
                self._drag_y = event.y_root

    def limpar(self):
        for w in self.inner.winfo_children():
            w.destroy()
        self.canvas.yview_moveto(0)

    # =========================================================================
    # CONTROLE DE TELAS E RENDERIZAÇÃO DO ESTADO
    # =========================================================================
    def _mostrar_tela(self, nome: str, frame: tk.Frame):
        if self.tela_ativa == nome:
            return
        for f in (
            self.frame_lobby,
            self.frame_distribuir_mao,
            self.frame_distribuir_mesa,
            self.frame_jogo,
            self.frame_virar_mesa,
            self.frame_showdown,
            self.frame_placar,
        ):
            f.place_forget()
        frame.place(x=0, y=ALTURA_HEADER, width=LARGURA, height=ALTURA_CORPO)
        frame.tkraise()
        self.header.tkraise()
        self.tela_ativa = nome

    def _processar_fila_ui(self):
        atualizar = False
        while not self.fila_ui.empty():
            try:
                self.fila_ui.get_nowait()
                atualizar = True
            except queue.Empty:
                break
        if atualizar:
            self._renderizar_estado()
        self.root.after(60, self._processar_fila_ui)

    def _renderizar_estado(self):
        s = self.servidor.obter_snapshot()
        fase = s["fase"]

        if fase == "AGUARDANDO":
            self.lbl_header_titulo.config(text=f"LOBBY • {self.ip_local}:5000")
            self._mostrar_tela("LOBBY", self.frame_lobby)
            self._atualizar_lobby(s)

        elif fase == "DISTRIBUINDO_MAOS":
            self.lbl_header_titulo.config(
                text=f"RODADA #{s['numero_rodada']} • DISTRIBUIR MÃOS"
            )
            self._mostrar_tela("DIST_MAO", self.frame_distribuir_mao)
            self._atualizar_distribuir_mao(s)

        elif fase == "DISTRIBUINDO_MESA":
            self.lbl_header_titulo.config(
                text=f"RODADA #{s['numero_rodada']} • CARTAS DA MESA"
            )
            self._mostrar_tela("DIST_MESA", self.frame_distribuir_mesa)
            self._atualizar_distribuir_mesa(s)

        elif fase in ("PRE_FLOP", "FLOP", "TURN", "RIVER"):
            nomes_fase = {
                "PRE_FLOP": "PRÉ-FLOP",
                "FLOP": "FLOP",
                "TURN": "TURN",
                "RIVER": "RIVER",
            }
            self.lbl_header_titulo.config(
                text=f"RODADA #{s['numero_rodada']} • {nomes_fase[fase]} • Mesa R${fmt_valor(s['montante_total'])}"
            )
            self._mostrar_tela("JOGO", self.frame_jogo)
            self._atualizar_jogo(s)

        elif fase in (
            "AGUARDANDO_VIRAR_FLOP",
            "AGUARDANDO_VIRAR_TURN",
            "AGUARDANDO_VIRAR_RIVER",
        ):
            self.lbl_header_titulo.config(
                text=f"RODADA #{s['numero_rodada']} • AÇÃO DO DEALER"
            )
            self._mostrar_tela("VIRAR", self.frame_virar_mesa)
            self._atualizar_virar_mesa(s)

        elif fase == "SHOWDOWN":
            self.lbl_header_titulo.config(
                text=f"RODADA #{s['numero_rodada']} • FINALIZADA"
            )
            self._mostrar_tela("SHOWDOWN", self.frame_showdown)
            self._atualizar_showdown(s)

class HardwareScanner:
    def __init__(
        self, pino_led: int, pino_vcc_ir: int, pino_do_ir: int, ativo_high: bool
    ):
        self.ativo = False
        try:
            self.led = DigitalOutputDevice(pino_led, initial_value=False)
            self.vcc_ir = DigitalOutputDevice(pino_vcc_ir, initial_value=False)
            self.sensor_ir = DigitalInputDevice(
                pino_do_ir, pull_up=None, active_state=ativo_high
            )
            self.ativo = True
            log("Hardware GPIO inicializado (LED, VCC IR e DO IR).")
        except Exception as e:
            log(f"Hardware GPIO indisponível ({e}); operando em fallback.")

    def ligar(self) -> None:
        if self.ativo:
            self.led.on()
            self.vcc_ir.on()
            time.sleep(0.05)

    def desligar(self) -> None:
        if self.ativo:
            self.led.off()
            self.vcc_ir.off()

    @property
    def carta_no_sensor(self) -> bool:
        if not self.ativo:
            return True
        return self.sensor_ir.is_active

    def fechar(self) -> None:
        if self.ativo:
            self.desligar()
            self.led.close()
            self.vcc_ir.close()
            self.sensor_ir.close()


def orientar(bruto):

def extrair_simbolo(recorte):
    cinza = cv2.cvtColor(recorte, cv2.COLOR_BGR2GRAY)
    if cv2.mean(cinza)[0] < LIMIAR_PRESENCA:
        return None

    cinza_blur = cv2.GaussianBlur(cinza, (3, 3), 0)
    _, binaria = cv2.threshold(
        cinza_blur, 0, 255, cv2.THRESH_BINARY_INV | cv2.THRESH_OTSU
    )

    contornos, _ = cv2.findContours(binaria, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    area_max = 0.5 * recorte.size / 3
    x_min, y_min, x_max, y_max = 9999, 9999, -1, -1
    achou = False

    for c in contornos:
        a = cv2.contourArea(c)
        if a < AREA_MIN or a > area_max:
            continue
        x, y, w, h = cv2.boundingRect(c)
        x_min = min(x_min, x)
        y_min = min(y_min, y)
        x_max = max(x_max, x + w)
        y_max = max(y_max, y + h)
        achou = True

    if not achou:
        return None

    tinta = binaria[y_min:y_max, x_min:x_max]
    lado_sq = max(tinta.shape[1], tinta.shape[0])
    margem = max(2, lado_sq // 8)

    quadrado = np.zeros((lado_sq + 2 * margem, lado_sq + 2 * margem), dtype=np.uint8)
    ox = margem + (lado_sq - tinta.shape[1]) // 2
    oy = margem + (lado_sq - tinta.shape[0]) // 2
    quadrado[oy : oy + tinta.shape[0], ox : ox + tinta.shape[1]] = tinta

    img_redim = cv2.resize(quadrado, (LADO, LADO), interpolation=cv2.INTER_AREA)

    mascara = binaria[y_min:y_max, x_min:x_max]
    recorte_roi = recorte[y_min:y_max, x_min:x_max]
    media_bgr = cv2.mean(recorte_roi, mask=mascara)
    vermelho = (media_bgr[2] - 0.5 * (media_bgr[0] + media_bgr[1])) > LIMIAR_VERMELHO

    return img_redim, vermelho


def prever(modelo, nomes, img) -> str:
    caracteristicas = np.float32(calcular_hog(img))
    _, r = modelo.predict(caracteristicas)
    i = int(r[0][0])
    return nomes[i] if 0 <= i < len(nomes) else "?"


def ler_comandos() -> None:
    for linha in sys.stdin:
        cmd = linha.strip()
        if not cmd:
            continue
        with lock_estado:
            if cmd == "LIGAR":
                estado["quer_camera"] = True
            elif cmd == "DESLIGAR":
                estado["quer_camera"] = False
            elif cmd == "RESET_LEITURA":
                estado["resetar_leitura"] = True
            elif cmd.startswith("USADAS;"):
                corpo = cmd.split(";", 1)[1].strip()
                estado["cartas_usadas"] = {
                    c.strip() for c in corpo.split(",") if c.strip()
                }
            elif cmd == "SAIR":
                estado["rodando"] = False
                break
            else:
                log(f"Comando desconhecido: {cmd}")
    with lock_estado:
        estado["rodando"] = False

                time.sleep(0.1)
                continue

            carta_presente = hw.carta_no_sensor

            if carta_presente != estado_ir_anterior:
                if carta_presente:
                    momento_deteccao_ir = time.monotonic()
                    log("[IR] Carta entrou no scanner. Aguardando estabilizar...")
                else:
                    if estado_ir_anterior is not None:
                        log("[IR] Carta removida do scanner.")
                        evento("CARTA_REMOVIDA")
                candidata_atual = None
                votos = 0
                votos_duplicada = 0
                carta_confirmada = False
                duplicada_notificada = None
                estado_ir_anterior = carta_presente

            if not carta_presente or carta_confirmada:
                time.sleep(0.03)
                continue

            if (time.monotonic() - momento_deteccao_ir) < TEMPO_ESTABILIZACAO:
                time.sleep(0.02)
                continue

            try:
                bruto = cap.capture_array("main")
            except Exception as e:
                log(f"ERRO ao capturar frame: {e}")
                time.sleep(0.05)
                continue

            frame = orientar(bruto)
            rec_num = frame[ry : ry + rh, rx : rx + rw]
            rec_nai = frame[ny : ny + nh, nx : nx + nw]

            res_v = extrair_simbolo(rec_num)
            res_n = extrair_simbolo(rec_nai)

            valor = (
                prever(svm_valores, NOMES_VALORES, res_v[0])
                if res_v is not None
                else "?"
            )
            naipe = "?"
            if res_n is not None:
                img_n, is_red = res_n
                naipe = (
                    prever(svm_vermelhos, NOMES_NAIPES_VERMELHOS, img_n)
                    if is_red
                    else prever(svm_pretos, NOMES_NAIPES_PRETOS, img_n)
                )

            if valor != "?" and naipe != "?":
                codigo = traduzir_codigo(valor, naipe)
                leitura = (valor, naipe)

                # Bloqueio inteligente de carta já usada na rodada:
                # descarta falso positivo momentâneo; só avisa duplicata se persistir
                if codigo in cartas_usadas:
                    votos = 0
                    if leitura == candidata_atual:
                        votos_duplicada += 1
                    else:
                        candidata_atual = leitura
                        votos_duplicada = 1

                    if (
                        votos_duplicada >= VOTOS_NECESSARIOS
                        and duplicada_notificada != codigo
                    ):
                        duplicada_notificada = codigo
                        log(f"[SCANNER] Carta duplicada bloqueada: {valor} de {naipe}")
                        evento(f"DUPLICADA;{valor};{naipe}")
                else:
                    votos_duplicada = 0
                    duplicada_notificada = None
                    if leitura == candidata_atual:
                        votos += 1
                    else:
                        candidata_atual = leitura
                        votos = 1

                    if votos >= VOTOS_NECESSARIOS:
                        carta_confirmada = True
                        log(f"[SCANNER] Carta confirmada: {valor} de {naipe}")
                        evento(f"CARTA;{valor};{naipe}")
            else:
                candidata_atual = None
                votos = 0
                votos_duplicada = 0

            time.sleep(0.03)
    finally:
        if cap is not None:

        self._bot_agendado = False

        self.roteador = {
            "registro": lambda conn, d: self.jogo.registrar_jogador(

        }

    # =========================================================================
    # CICLO DE VIDA (INICIAR / ENCERRAR COM SEGURANÇA)
    # =========================================================================
    def iniciar(self) -> None:
        self._rodando = True
        threading.Thread(target=self._loop_sockets, daemon=True).start()
        self._iniciar_scanner()
        self._notificar_ui()

    def encerrar(self) -> None:
        """Desliga LED/IR/câmera no scanner e fecha todos os sockets com segurança."""
        self._rodando = False

        # 1. Encerra o processo do scanner (desliga GPIO LED/IR e fecha câmera)
        if self._processo_scanner is not None:
            try:
                if self._processo_scanner.stdin:
                    self._processo_scanner.stdin.write(b"SAIR\n")
                    self._processo_scanner.stdin.flush()
                self._processo_scanner.wait(timeout=3)
            except Exception:
                try:
                    self._processo_scanner.terminate()
                except Exception:
                    pass
            self._processo_scanner = None

        # 2. Fecha sockets de clientes e servidor
        with self.lock:
            for jid in list(self.jogo.jogadores.keys()):
                if isinstance(jid, socket.socket):
                    try:
                        jid.close()
                    except Exception:
                        pass

        if self._sock_servidor is not None:
            try:
                self._sock_servidor.close()
            except Exception:
                pass
            self._sock_servidor = None

    # =========================================================================
    # COMUNICAÇÃO COM O SUBPROCESSO DO SCANNER
    # =========================================================================
    def _iniciar_scanner(self) -> None:
        script_scanner = Path(__file__).resolve().parent / "scanner_integrado.py"
        try:
            self._processo_scanner = subprocess.Popen(
                [sys.executable, str(script_scanner)],
                stdin=subprocess.PIPE,
                stdout=subprocess.PIPE,
                stderr=sys.stderr,
            )
            threading.Thread(target=self._loop_scanner, daemon=True).start()
        except Exception as e:
            print(f"[Servidor] Falha ao iniciar scanner_integrado.py: {e}")

    def _enviar_cmd_scanner(self, cmd: str) -> None:
        proc = self._processo_scanner
        if proc is not None and proc.poll() is None and proc.stdin is not None:
            try:
                proc.stdin.write(f"{cmd}\n".encode("utf-8"))
                proc.stdin.flush()
            except Exception:
                pass

    def _sincronizar_scanner_com_fase(self) -> None:
        """Liga câmera/LED apenas na fase de escaneamento e sincroniza cartas usadas."""
        precisa_scanner = self.jogo.fase in ("DISTRIBUINDO_MAOS", "DISTRIBUINDO_MESA")
        if precisa_scanner and not self._camera_ligada:
            self._camera_ligada = True
            self._enviar_cmd_scanner("LIGAR")
        elif not precisa_scanner and self._camera_ligada:
            self._camera_ligada = False
            self._enviar_cmd_scanner("DESLIGAR")

        lista_usadas = ",".join(sorted(self.jogo.cartas_usadas_rodada))
        self._enviar_cmd_scanner(f"USADAS;{lista_usadas}")

    def _loop_scanner(self) -> None:
        proc = self._processo_scanner
        if proc is None or proc.stdout is None:
            return
        for linha in proc.stdout:
            if not self._rodando:
                break
            evento = linha.decode("utf-8", errors="ignore").strip()
            if not evento:
                continue

            print(f"[Scanner] {evento}")
            if evento.startswith("CARTA;"):
                partes = evento.split(";")
                if len(partes) == 3:
                    carta = traduzir_carta(partes[1], partes[2])
                    with self.lock:
                        msgs = self.jogo.processar_carta_lida(carta)
                        self._sincronizar_scanner_com_fase()
                    self._despachar(msgs)
                    self._notificar_ui()

            elif evento.startswith("DUPLICADA;"):
                with self.lock:
                    self.jogo.registrar_aviso_duplicada()
                self._notificar_ui()

    # =========================================================================
    # REDE TCP (CELULARES ANDROID)
    # =========================================================================
    def _loop_sockets(self) -> None:
        srv = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        srv.bind(("0.0.0.0", self.porta))
        srv.listen()
        srv.settimeout(1.0)
        self._sock_servidor = srv
        print(f"[Rede] Servidor Poker rodando na porta {self.porta}...")

        while self._rodando:
            try:
                conn, addr = srv.accept()
            except socket.timeout:
                continue
            except OSError:
                break
            threading.Thread(
                target=self._atender_cliente, args=(conn, addr), daemon=True
            ).start()

    def _atender_cliente(self, conn: socket.socket, addr) -> None:
        print(f"[Rede] Celular conectado: {addr}")
        try:
            conn.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        except Exception:
            pass
        buffer = ""
        try:
            while self._rodando:
                dados_brutos = conn.recv(1024).decode("utf-8")
                if not dados_brutos:
                    break
                buffer += dados_brutos
                while "\n" in buffer:
                    linha, buffer = buffer.split("\n", 1)
                    if not linha.strip():
                        continue
                    tipo, dados = desempacotar(linha)
                    handler = self.roteador.get(tipo)
                    if handler:
                        with self.lock:
                            msgs = handler(conn, dados)
                            self._sincronizar_scanner_com_fase()
                            precisa_bot = self.jogo.jogador_da_vez_eh_bot()
                        self._despachar(msgs)
                        self._notificar_ui()
                        if precisa_bot:
                            self._agendar_jogada_bot()
        except Exception as e:
            print(f"[Rede] Erro com cliente {addr}: {e}")
        finally:
            with self.lock:
                msgs = self.jogo.remover_jogador(conn)
                self._sincronizar_scanner_com_fase()
                precisa_bot = self.jogo.jogador_da_vez_eh_bot()
            self._despachar(msgs)
            self._notificar_ui()
            if precisa_bot:
                self._agendar_jogada_bot()
            try:
                conn.close()
            except Exception:
                pass
            print(f"[Rede] Celular desconectado: {addr}")

    def _despachar(self, mensagens: list[tuple]) -> None:
        if not mensagens:
            return
        with self.lock:
            sockets_humanos = [
                jid
                for jid in self.jogo.jogadores.keys()
                if isinstance(jid, socket.socket)
            ]

        lotes_por_conn: dict[socket.socket, list[bytes]] = {}
        for destino, tipo, dados in mensagens:
            payload = empacotar(tipo, **dados)
            if destino == "TODOS":
                destinatarios = sockets_humanos
            elif isinstance(destino, socket.socket):
                destinatarios = [destino]
            else:
                continue

            for conn in destinatarios:
                lotes_por_conn.setdefault(conn, []).append(payload)

        for conn, pacotes in lotes_por_conn.items():
            try:
                conn.sendall(b"".join(pacotes))
            except Exception:
                pass

    # =========================================================================
    # BOT DE TESTES (AGENDAMENTO ASSÍNCRONO)
    # =========================================================================
    def _agendar_jogada_bot(self) -> None:
        with self.lock:
            if self._bot_agendado or not self._rodando:

            self._bot_agendado = True

        def executar():
            time.sleep(1.1)

            if novo_turno_bot:
                self._agendar_jogada_bot()

        threading.Thread(target=executar, daemon=True).start()

import json


def empacotar(tipo: str, **dados) -> bytes:
    """Cria uma mensagem padronizada no formato {'tipo': ..., 'dados': {...}}\n."""
    envelope = {"tipo": tipo, "dados": dados}
    return (json.dumps(envelope) + "\n").encode("utf-8")


def desempacotar(linha: str) -> tuple[str, dict]:
    """Extrai (tipo, dados) de uma linha JSON recebida pelo socket."""
    envelope = json.loads(linha)
    return envelope.get("tipo", ""), envelope.get("dados", {})

class AndroidTTS:
    """Wrapper para o Text-to-Speech nativo do Android via Pyjnius."""

    def __init__(self):
        self.ativo = True
        self._tts = None
        self._idioma_configurado = False
        self._queue_add = 1
        self._queue_flush = 0
        self._utterance_id = 0
        self._lock = threading.Lock()
        self._inicializar_android()

    def _inicializar_android(self):
        try:
            from jnius import autoclass

            ActivityThread = autoclass("android.app.ActivityThread")
            TextToSpeech = autoclass("android.speech.tts.TextToSpeech")

            app = ActivityThread.currentApplication()
            if app is None:
                return
            context = app.getApplicationContext()

            self._queue_add = TextToSpeech.QUEUE_ADD
            self._queue_flush = TextToSpeech.QUEUE_FLUSH
            self._tts = TextToSpeech(context, None)
            self._garantir_idioma()
        except Exception as e:
            print(f"[TTS] Modo desktop/fallback: {e}")
            self._tts = None

    def _garantir_idioma(self):
        if self._tts is not None and not self._idioma_configurado:
            self._idioma_configurado = True
            try:
                from jnius import autoclass

                Locale = autoclass("java.util.Locale")
                res = self._tts.setLanguage(Locale("pt", "BR"))
                if res < 0:
                    self._tts.setLanguage(Locale.getDefault())
            except Exception:
                pass

    def falar(self, texto: str, interromper: bool = False):
        texto = texto.strip()
        if not texto or not self.ativo:
            return

        with self._lock:
            self._utterance_id += 1
            uid = f"poker_tts_{self._utterance_id}"

        if self._tts is not None:
            try:
                self._garantir_idioma()
                modo = self._queue_flush if interromper else self._queue_add
                self._tts.speak(texto, modo, None, uid)
            except Exception as e:
                print(f"[TTS] Erro ao falar: {e}")
        else:
            print(f"[TTS] {texto}")

    def parar(self):
        if self._tts is not None:
            try:
                self._tts.stop()
            except Exception:
                pass


def carta_para_texto_tts(codigo: str) -> str:
    if len(codigo) < 2:
        return codigo
    valor = NOMES_VALORES_TTS.get(codigo[:-1].upper(), codigo[:-1])

def formatar_acao_para_tts(msg: str) -> str | None:
    """Filtra eventos de jogo e adapta o texto para pronúncia natural em português."""
    if not msg:
        return None

    palavras_chave = (
        "correu",
        "passou",
        "pagou",
        "aumentou",
        "ALL-IN",
        "ganhou",
        "SHOWDOWN",
        "Blinds:",
        "Flop virado",
        "Turn virado",
        "River virado",
        "Rebuy",
        "conectou",
        "entrou",
        "saiu",
        "desconectou",
    )
    if not any(k in msg for k in palavras_chave):
        return None

    texto = msg
    for termo in (" (Fold)", " (Check)", " (Call)", " (Raise)"):
        texto = texto.replace(termo, "")
    texto = texto.replace("ALL-IN", "all-in").replace("SHOWDOWN!", "Showdown!")
    texto = texto.replace("(SB ", "(small blind ").replace("(BB ", "(big blind ")
    texto = re.sub(r"R\$\s*(\d+(?:[.,]\d+)?)", r"\1 reais", texto)
    return texto

def _avaliar_5(cinco: tuple[str, ...]) -> tuple[int, list[int]]:
    valores = sorted([VALORES_MAP[c[0].upper()] for c in cinco], reverse=True)
    naipes = [c[1].lower() for c in cinco]
    eh_flush = len(set(naipes)) == 1
    eh_straight = (valores == list(range(valores[0], valores[0] - 5, -1))) or (
        valores == [14, 5, 4, 3, 2]
    )
    if valores == [14, 5, 4, 3, 2]:
        valores = [5, 4, 3, 2, 1]

    contagem = sorted([(valores.count(v), v) for v in set(valores)], reverse=True)
    freq = [p[0] for p in contagem]
    desempate = [p[1] for p in contagem]

    if eh_straight and eh_flush:
        return (8, desempate)
    if freq == [4, 1]:
        return (7, desempate)
    if freq == [3, 2]:
        return (6, desempate)
    if eh_flush:
        return (5, desempate)
    if eh_straight:
        return (4, desempate)
    if freq == [3, 1, 1]:
        return (3, desempate)
    if freq == [2, 2, 1]:
        return (2, desempate)
    if freq == [2, 1, 1, 1]:
        return (1, desempate)
    return (0, desempate)


def descobrir_combinacao(mao: list[str], mesa: list[str]) -> str:
    todas = [c for c in (mao + mesa) if len(c) >= 2 and c[0].upper() in VALORES_MAP]
    if not todas:
        return "Aguardando cartas..."
    if len(todas) < 5:
        valores = [VALORES_MAP[c[0].upper()] for c in todas]
        cont = sorted([valores.count(v) for v in set(valores)], reverse=True)
        if cont[0] == 4:
            return "Quadra"
        if cont[0] == 3:
            return "Trinca"
        if cont[:2] == [2, 2]:
            return "Dois Pares"
        if cont[0] == 2:
            return "Um Par"
        return "Carta Alta"
    melhor = max(_avaliar_5(comb) for comb in combinations(todas, 5))
    return NOMES_COMBINACOES[melhor[0]]

    def escutar_servidor():
        nonlocal conectado
        buffer = ""
        try:
            while conectado and sock_cliente:
                dados_brutos = sock_cliente.recv(4096).decode("utf-8")
                if not dados_brutos:
                    break
                buffer += dados_brutos
                atualizou = False
                while "\n" in buffer:
                    linha, buffer = buffer.split("\n", 1)
                    if not linha.strip():
                        continue
                    try:
                        tipo, dados = desempacotar(linha)
                        handler = handlers_eventos.get(tipo)
                        if handler:
                            handler(dados)
                            atualizou = True
                    except Exception as ui_err:
                        txt_info.value = f"Erro de interface: {ui_err}"
                        atualizou = True

                    page.update()
        except Exception as net_err:
            if conectado:
                voltar_para_conexao(f"Desconectado: {net_err}")
                return

            voltar_para_conexao("Servidor encerrou a conexão.")

        nome = input_nome.value.strip()

            notificar_erro_login("Preencha o IP do servidor e o seu Nome.")

        try:
            porta = int(input_porta.value.strip())
            if not (1 <= porta <= 65535):
                raise ValueError
        except ValueError:
            notificar_erro_login("Porta inválida. Digite um número entre 1 e 65535.")
            return

        try:
            saldo_inicial = float(input_saldo.value.strip().replace(",", "."))
        except ValueError:
            notificar_erro_login("Saldo inicial inválido. Digite apenas números.")
            return

            notificar_erro_login(
                f"Saldo Inicial deve ser entre R$ {fmt_valor(SALDO_MIN)} e R$ {fmt_valor(SALDO_MAX)}."
            )
            return

        btn_conectar.disabled = True
        btn_conectar.content = "Conectando..."
        box_notificacao.visible = False
        page.update()

        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5.0)
            sock.connect((ip, porta))
            sock.settimeout(None)

            sock_cliente = sock
            conectado = True

            enviar_comando("registro", nome=nome, saldo=saldo_inicial)
            ir_para_tela_mesa(nome, saldo_inicial, ip, porta)

            threading.Thread(target=escutar_servidor, daemon=True).start()
        except Exception as ex:
            btn_conectar.disabled = False
            btn_conectar.content = "Entrar na Mesa"
            notificar_erro_login(f"Não foi possível conectar em {ip}:{porta} ({ex})")
